cmake_minimum_required(VERSION 3.28)
project(PolarNative VERSION 0.2.0 LANGUAGES C CXX)
if(MSVC)
  get_filename_component(POLAR_COMPILER_DIRECTORY "${CMAKE_CXX_COMPILER}" DIRECTORY)
  if(NOT EXISTS "${POLAR_COMPILER_DIRECTORY}/1033/clui.dll")
    message(FATAL_ERROR "Install the en-US Visual Studio Build Tools language pack for reliable Ninja header dependencies")
  endif()
  # Keep Ninja's header dependency scanner independent of the host UI language.
  set(CMAKE_C_COMPILER_LAUNCHER "${CMAKE_COMMAND};-E;env;VSLANG=1033")
  set(CMAKE_CXX_COMPILER_LAUNCHER "${CMAKE_COMMAND};-E;env;VSLANG=1033")
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "Note: including file: ")
endif()
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Polar Native currently targets Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(POLAR_ASAN "Address sanitizer" OFF)
option(POLAR_GUI "Build native GUI example" ON)
if(POLAR_ASAN)
 add_compile_options(/fsanitize=address /Zi)
 add_link_options(/INCREMENTAL:NO)
endif()
include(cmake/Dependencies.cmake)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/resources/polar_device_capabilities.json" POLAR_CAPABILITIES_JSON)
configure_file(native/capability_config.hpp.in capability_config.hpp @ONLY)
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
file(GLOB POLAR_PROTO_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/proto/*.proto")
add_library(polar_proto STATIC ${POLAR_PROTO_FILES})
target_link_libraries(polar_proto PUBLIC protobuf::libprotobuf)
target_include_directories(polar_proto PUBLIC "${CMAKE_CURRENT_BINARY_DIR}")
protobuf_generate(TARGET polar_proto IMPORT_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/proto" "${protobuf_SOURCE_DIR}/src" APPEND_PATH)
add_library(polar_core STATIC native/decoder.cpp native/codecs.cpp native/services.cpp native/device_services.cpp native/control_services.cpp native/capabilities.cpp)
target_include_directories(polar_core PUBLIC include native)
target_compile_options(polar_core PRIVATE /W4 /permissive- /EHsc /bigobj)
target_link_libraries(polar_core PUBLIC polar_proto ZLIB::ZLIB MINIZIP::minizip bcrypt winhttp)
add_library(polar_sdk SHARED native/sdk.cpp native/windows_ble.cpp)
target_compile_definitions(polar_sdk PRIVATE POLAR_SDK_BUILD)
target_compile_options(polar_sdk PRIVATE /W4 /permissive- /EHsc /bigobj)
target_link_libraries(polar_sdk PRIVATE polar_core windowsapp runtimeobject)
set_property(TARGET polar_sdk PROPERTY LINK_LIBRARY_OVERRIDE_polar_proto WHOLE_ARCHIVE)
target_include_directories(polar_sdk PUBLIC include)
set_target_properties(polar_sdk PROPERTIES OUTPUT_NAME polar_sdk)
add_executable(polar_diag tools/diagnostic.cpp)
target_link_libraries(polar_diag PRIVATE polar_sdk)
add_executable(polar_replay tools/replay.cpp)
target_link_libraries(polar_replay PRIVATE polar_sdk)
add_executable(polar_benchmark tools/benchmark.cpp)
target_link_libraries(polar_benchmark PRIVATE polar_core)
enable_testing()
add_executable(polar_tests tests/native/tests.cpp tests/native/service_tests.cpp tests/native/lifetime_tests.cpp)
target_link_libraries(polar_tests PRIVATE polar_sdk polar_core)
set_property(TARGET polar_tests PROPERTY LINK_LIBRARY_OVERRIDE_polar_proto WHOLE_ARCHIVE)
add_test(NAME polar_native COMMAND polar_tests)
set_tests_properties(polar_native PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(polar_engine_tests tests/native/engine_tests.cpp native/sdk.cpp)
target_compile_definitions(polar_engine_tests PRIVATE POLAR_SDK_BUILD)
target_compile_options(polar_engine_tests PRIVATE /EHsc /bigobj)
target_link_libraries(polar_engine_tests PRIVATE polar_core runtimeobject windowsapp)
add_test(NAME polar_engine_lifetime COMMAND polar_engine_tests)
if(POLAR_GUI)
 add_executable(polar_example WIN32 samples/windows/main.cpp ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_win32.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_dx11.cpp)
 target_include_directories(polar_example PRIVATE ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
 target_link_libraries(polar_example PRIVATE polar_sdk d3d11 dxgi dwmapi)
endif()
install(TARGETS polar_sdk polar_diag polar_replay polar_benchmark RUNTIME DESTINATION bin ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY proto/ DESTINATION schemas)
install(DIRECTORY LICENSES/ DESTINATION licenses)
install(FILES NOTICE README.md DESTINATION .)
install(FILES cmake/PolarNativeConfig.cmake DESTINATION lib/cmake/PolarNative)
install(DIRECTORY samples/consumer/ DESTINATION examples/consumer)
install(DIRECTORY docs/ DESTINATION docs)
install(DIRECTORY resources/ DESTINATION resources)
